	\section{Analyse théorique des dynamiques apparentes}
	\label{app_apparent_dyn_robot}
	
	%\paragraph{Analyse théorique} 
	
	On donne le modèle dynamique général d'un robot en contact avec un environnement \cref{eq_dyn_model_robot}, avec $ \matr{M} $ la matrice d'inertie du robot, $ \vect{h} $ le vecteur regroupant les phénomènes liées à la position et à la vitesse - \textit{e.g., couples centrifuges, de Coriolis, gravitationnel}, et $ \vect{\tau}_r $ et $ \vect{\tau}_{e} $ les couples articulaires respectivement générés par les actionneurs du robot et par l'environnement extérieur. Ce modèle dynamique peut-être obtenu à partir des équations de Lagrange, comme explicité par \citet[Chapitre 9, \textit{Dynamic modeling of serial robots}]{w_khalil_modeling_2002}.
	%
	\begin{equation} \label{eq_dyn_model_robot}
	\matr{M}(\vect{q})\ddot{\vect{q}} + \vect{h}(\dot{\vect{q}}, \vect{q}) = \vect{\tau}_r + \vect{\tau}_{e}
	\end{equation}
	%
	En explicitant les termes regroupés par $ \vect{h} $, on peut écrire l'expression suivante qui sépare les effets inertiels de ceux liés à la gravité $ \vect{g}(\vect{q}) $. En considérant les phénomènes de frottements visqueux notés $ \matr{F}_v $ ainsi que les couples centrifuges et de Coriolis regroupés dans le terme $ \matr{c}(\dot{\vect{q}}, \vect{q}) $ aussi détaillé par \cite{w_khalil_modeling_2002}, on peut exprimer l'accélération comme une fonction $ \vect{f} $ dépendant des positions et vitesses articulaires, respectivement  $ \vect{q}, \dot{\vect{q}} $, des couples internes $ \vect{\tau}_r $ et des forces générés par l'environnement $ \vect{f}_e $.
	%
	\begin{equation} \label{eq_dyn_model_robot_complete}
		\ddot{\vect{q}} = \matr{M}(\vect{q})^{-1} \left( \vect{\tau}_r + \jacobian{\vect{q}
		}{}{t} \vect{f}_{e} - \matr{F}_v \dot{\vect{q}} - \matr{C}(\vect{q}, \dot{\vect{q}}) \dot{\vect{q}} - \vect{g}(\vect{q}) \right) = \vect{f}(\vect{q}, \dot{\vect{q}}, \vect{\tau}_r, \vect{f}_e)
	\end{equation}
	%
	\subsection{Linéarisation autour de la configuration nominale}
	%
	En s'intéressant à de mouvements faibles autours de la configuration  $ \vect{q}_0 $, on peut exprimer $ d\vect{f}( \vect{q}, \dot{\vect{q}}, \vect{\tau}_r, \vect{\tau}_e) $ comme la linéarisation du modèle décrit en \cref{eq_dyn_model_robot_complete}, autours des points d'équilibre $ \matr{X}_0 = \left( \vect{q}_0, \dot{\vect{q}_0}, \vect{\tau}_{r0}, \vect{f}_{e0} \right) $, avec $ \dot{\vect{q}}_0 = \vect{f}_{e0} = 0 $.
	%
	\begin{equation} \label{eq_linearization_dyn}
		d\ddot{\vect{q}} = \left. \frac{\partial \vect{f}}{\partial \vect{q} } \right|_{\matr{X}_0} d\vect{q} + \left. \frac{\partial \vect{f}}{\partial \dot{\vect{q}} } \right|_{\matr{X}_0} d\dot{\vect{q}} + \left. \frac{\partial \vect{f}}{\partial \vect{\tau_r}} \right|_{\matr{X}_0} d\vect{\tau_r} + \left. \frac{\partial \vect{f}}{\partial \vect{f_e}} \right|_{\matr{X}_0} d\vect{f_e}
	\end{equation}
	%
	Si on s'intéresse à la première dérivée partielle, en fonction de $ \vect{q} $, on trouve des termes qui vont s'annuler à l'équilibre, d'abord explicité en \cref{eq_linearization_1st_partial}.
	\begin{equation} \label{eq_linearization_1st_partial}
		\frac{\partial \vect{f}}{\partial \vect{q} } = \frac{\partial \matr{M}(\vect{q})^{-1} \vect{\tau}_{r}}{\partial \vect{q} } + \frac{\partial \matr{M}(\vect{q})^{-1} \jacobian{\vect{q}}{}{t} \vect{f}_{e}}{\partial \vect{q} } - \frac{\partial \matr{M}(\vect{q})^{-1} \left( \matr{F}_v + \matr{C}(\vect{q}, \dot{\vect{q}}) \right) \dot{\vect{q}}}{\partial \vect{q} } - \frac{\partial \matr{M}(\vect{q})^{-1} \vect{g}(\vect{q})}{\partial \vect{q} } 
	\end{equation}
	Les dérivées partielles sont ensuite exprimées à l'équilibre, ce qui donne les matrices de coefficients exprimés ci-après en \cref{eq_linearization_cst_terms}.
	\begin{equation} \label{eq_linearization_cst_terms}
	\begin{split}
		\left. \frac{\partial \vect{f}}{\partial \vect{q} } \right|_{\matr{X}_0} &= \left. \frac{\partial \matr{M}(\vect{q})^{-1} \vect{\tau}_{r0}}{\partial \vect{q} } \right|_{\vect{q}_0} - \left. \frac{\partial \matr{M}(\vect{q})^{-1} \vect{g}(\vect{q})}{\partial \vect{q} } \right|_{\vect{q}_0} = \matr{K}^\dagger(\vect{q}_0, \vect{\tau}_{r0}) \\
		\left. \frac{\partial \vect{f}}{\partial \dot{\vect{q}} } \right|_{\matr{X}_0} &= -\matr{M}(\vect{q}_0)^{-1}\matr{F}_v - \matr{M}(\vect{q}_0)^{-1} \left. \frac{\partial \matr{C}(\vect{q}, \dot{\vect{q}}) \dot{\vect{q}}}{\partial \dot{\vect{q}} } \right|_{\matr{X}_0} = \matr{M}(\vect{q}_0)^{-1} \matr{V}^\dagger(\vect{q}_0)\\
		\left. \frac{\partial \vect{f}}{\partial \vect{\tau}_r} \right|_{\matr{X}_0} &= \matr{M}(\vect{q}_0)^{-1} \\		
		\left. \frac{\partial \vect{f}}{\partial \vect{f}_e} \right|_{\matr{X}_0} &= \matr{M}(\vect{q}_0)^{-1} \jacobian{\vect{q}_0}{}{t} 	
	\end{split}
	\end{equation}
	%
	\begin{equation} \label{eq_linearization_dyn_complete}
		d\ddot{\vect{q}} = \matr{K}^\dagger(\vect{q}_0, \vect{\tau}_{r0})  d\vect{q} + %
		%
		\matr{M}(\vect{q}_0)^{-1} \left( \matr{V}^\dagger(\vect{q}_0) d\dot{\vect{q}} + %
		%
		d\vect{\tau_r} + %
		%
		\jacobian{\vect{q}_0}{}{t} d\vect{f_e} \right)
	\end{equation}

	\subsubsection{Passage dans le plan cartésien}
	%
	En utilisant la relation \cref{eq_link_joint_endpoint_velocity} pour passer des variables articulaires à des variables cartésiennes, et en notant que la dérivée est donnée par \cref{eq_link_joint_endpoint_acceleration}, on peut alors écrire l'équation \cref{eq_dyn_model_robot_2}.
	%
	\begin{equation} \label{eq_link_joint_endpoint_velocity}
	\begin{split}
		&d\vect{x} = \jacobian{\vect{q}}{}{} d\vect{q} \\
		&\dot{\vect{X}} = \jacobian{\vect{q}}{}{} \dot{\vect{q}}
	\end{split}
	\end{equation}
	%
	\begin{equation} \label{eq_link_joint_endpoint_acceleration}
	\ddot{\vect{X}} = \jacobian{\vect{q}}{}{} \ddot{\vect{q}} + \jacobianDot{\vect{q}}{}{} \dot{\vect{q}}
	\end{equation}
	%
	On notera $ \jacobian{}{}{} = \jacobian{\vect{q}_0}{}{} $, et $ \matr{M} = \matr{M}(\vect{q}_0) $, pour simplifier l'écriture.
	%
	\begin{equation} \label{eq_dyn_model_robot_2}
		d\ddot{\vect{x}} - \jacobianDot{}{}{} \jacobian{}{}{-1} d\dot{\vect{x}} = \jacobian{}{}{} \matr{M}^{-1} \left( d\vect{\tau}_r + \jacobian{}{}{t} d\vect{f}_{e}  + \matr{V}^\dagger \jacobian{}{}{-1} d\dot{\vect{x}} \right) +  \jacobian{}{}{}\matr{K}^\dagger \jacobian{}{}{-1} d\vect{x}
	\end{equation}
	%
	\subsubsection{Intégration de la commande en couple du robot}
	%
	Il ne reste alors plus qu'à exprimer $ \vect{\tau}_r $ qui est donné par le contrôle en admittance, et la boucle interne en vitesses articulaires du robot, dont on note le correcteur $ \matr{C}_v \in \R^{3\times3} $. $ \matr{C}_v $ est diagonal. Les boucles secondaires, qui ne devraient pas avoir d'effet sur la transparence, sont négligées.
	%
	\begin{equation} \label{velocity_cmd_ctrl_loop}
		\dot{\vect{q}}_d = \jacobianHat{}{}{-1} \begin{pmatrix} 0 & c_z f_z & 0 \end{pmatrix}^{t} = \jacobianHat{}{}{-1} \matr{C}_z \vect{f}_e
	\end{equation}
	%
	\begin{equation} \label{torque_cmd_ctrl_loop}
		\vect{\tau}_r \approx \vect{\tau}_d = \matr{C}_v \left( \dot{\vect{q}}_d - \dot{\vect{q}} \right)
	\end{equation}
	%
	En intégrant \cref{velocity_cmd_ctrl_loop} et \cref{torque_cmd_ctrl_loop} dans \cref{eq_dyn_model_robot_2}, et en passant dans le domaine de Laplace, en notant $ \mathcal{L}\left(\dot{\vect{x}}(t)\right) = \vect{v}_e(s) $ et $ \mathcal{L}\left(\vect{f}_e(t)\right) = \vect{f}_e(s) $:
	%
%	\begin{equation} \label{eq_dyn_model_robot_admit_ctrl}
%		\ddot{\vect{x}} - \jacobianDot{}{}{} \jacobian{}{}{-1} \dot{\vect{x}} = \jacobian{}{}{} \matr{M}^{-1} \left( 
%		\matr{C}_v(s) \left( \jacobianHat{}{}{-1} \matr{C}_z(s) \vect{f}_e - \jacobian{}{}{-1} \dot{\vect{x}} \right)
%		+ \jacobian{}{}{t} \vect{f}_{e} - \matr{V}^\dagger \jacobian{}{}{-1} \dot{\vect{x}} \right) - \jacobian{}{}{}\matr{K}^\dagger \jacobian{}{}{-1} \vect{x}\text{,}
%	\end{equation}
	%
	\begin{equation} \label{eq_dyn_model_robot_admit_ctrl}
		s\vect{v}_e - \jacobianDot{}{}{} \jacobian{}{}{-1} \vect{v}_e = \jacobian{}{}{} \matr{M}^{-1} \left( 
		\matr{C}_v \left( \jacobianHat{}{}{-1} \matr{C}_z \vect{f}_e - \jacobian{}{}{-1} \vect{v}_e \right)
		+ \jacobian{}{}{t} \vect{f}_{e} + \matr{V}^\dagger \jacobian{}{}{-1} \vect{v}_e \right) + \jacobian{}{}{}\matr{K}^\dagger \jacobian{}{}{-1} \frac{\vect{v}_e}{s}\text{,}
	\end{equation}
	%
	que l'on peut réarranger pour exprimer la relation entre les forces externes $\vect{f}_{e} $ et la vitesse cartésienne de l'effecteur final :%, $ \vect{v}_{e} = s\vect{x}$:
	%
	\begin{equation} \label{eq_dyn_model_robot_imp_ctrl_2}
	\begin{split}
		\vect{v}_{e}(s) &= \matr{B}(s)^{-1}\matr{A}(s) \vect{f}_{e}(s) \\
		\matr{A}(s) &= \jacobian{}{}{} \matr{M}^{-1} \left( \matr{C}_v(s) \jacobianHat{}{}{-1} \matr{C}_z(s) + \jacobian{}{}{t} \right) \\
		\matr{B}(s) &= s\matr{I} + \left( \jacobian{}{}{} \matr{M}^{-1} \left( \matr{C}_v(s) - \matr{V}^\dagger \right) - \jacobianDot{}{}{}\right)\jacobian{}{}{-1} - \jacobian{}{}{} \matr{K}^\dagger \jacobian{}{}{-1} s^{-1} 
	\end{split}
	\end{equation}
	%
	\begin{equation}
	\begin{split}
	 \matr{C}_v(s) &= \matr{K}_v + \matr{K}_{vi} s^{-1} = 
	 \begin{pmatrix} 
		 	k_v^1 + k_{vi}^1 s^{-1} & 0 & 0 \\
		 	0 & k_v^2 + k_{vi}^2 s^{-1} & 0 \\
		 	0 & 0 &	k_v^3 + k_{vi}^3 s^{-1} \\
	 \end{pmatrix}\\
		\matr{C}_z(s) &= \matr{K}_p + \matr{K}_{i} s^{-1} = 
		\begin{pmatrix} 
		0 & 0 & 0 \\
		0 & c_z(s) & 0 \\
		0 & 0 &	0 \\
		\end{pmatrix} = 
		\begin{pmatrix} 
			0 & 0 & 0 \\
			0 & k + k_{i} s^{-1} & 0 \\
			0 & 0 &	0 \\
		\end{pmatrix}
	\end{split}
	\end{equation}
	%
	La linéarisation réalisée n'est valide que pour des dynamiques faibles (vitesse, accélération), on s'intéresse donc au comportement apparent en basse fréquence.
	%
	\begin{equation} \label{dynamic_apparent_lim0}
	\begin{split}
		\lim_{s \to 0} \vect{v}_{e}(s) &= \matr{B}(s \to 0)^{-1} \matr{A}(s \to 0) \vect{f}_{e}(s) \\
		\lim_{s \to 0} \matr{A}(s) &= \jacobian{}{}{} \matr{M}^{-1} \matr{K}_{vi} \jacobianHat{}{}{-1} \matr{K}_{i} s^{-2} \\
		\lim_{s \to 0} \matr{B}(s) &= \jacobian{}{}{} \left(\matr{M}^{-1} \matr{K}_{vi} - \matr{K}^\dagger  \right) \jacobian{}{}{-1} s^{-1}
	\end{split}
	\end{equation}
	%
	En supposant que l'estimation du jacobien est idéale, $ \jacobianHat{}{}{} = \jacobian{}{}{} $, l'expression \cref{dynamic_apparent_lim0} devient \cref{dynamic_apparent_lim0_2}, en notant l'exposant $+$ comme une pseudo-inverse.
	%
	\begin{equation} \label{dynamic_apparent_lim0_2}
	 	\lim_{s \to 0} \vect{v}_{e}(s) = s^{-1} \left( \jacobian{}{}{} \left( \matr{M}^{-1} \matr{K}_{vi} - \matr{K}^\dagger \right)^{-1} \matr{M}^{-1} \matr{K}_{vi} \jacobianHat{}{}{-1} \matr{K}_{i} \right) \vect{f}_{e}(s)
	\end{equation}	
	%
	Le résultat de \cref{dynamic_apparent_lim0_2} indique comme attendu \citep{lamy_achieving_2009}, que le comportement dynamique en basse fréquence s'apparente à celui d'une masse d'une valeur inversement proportionnelle au gain $ K_i $.
