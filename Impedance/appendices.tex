	Dans le cas très simple du robot plan, dont on rappelle la structure simplifiée, avec les notations des différentes longueurs et conventions d'angles en \cref{fig_robot_simplified_model}, le calcul du modèle géométrique direct ne nécessite pas de passer par la convention de Denavit-Hartenberg, et les combinaisons de transformations homogènes.
	
	\input{latex_figures/fig_robot_model}
	
	Ainsi, en cherchant à calculer géométriquement les coordonnées $ x_e $, $ z_e $ et $ \theta_e $, par rapport à la base du robot $ ( x_0, 0 ) $ on obtient les résultats présentés dans l'équation \cref{eq_direct_kinematic}.
	
	\begin{equation} \label{eq_direct_kinematic}
		\begin{split}
			x_e &= (x_0 - a) + l_1\cos(q_1) + l_2\cos(q_1 + q_2) + l_3\cos(q_1 + q_2 + q_3) \\
			z_e &= l_0 + l_1\sin(q_1) + l_2\sin(q_1 + q_2) + l_3\sin(q_1 + q_2 + q_3) \\
			\theta_e &= q_1 + q_2 + q_3
		\end{split}
	\end{equation}

	\section{Calcul du Jacobien} \label{app_jacobian}
	
	\subsection{Jacobien du modèle réduit} \label{app_jacobian_sub_3}
	
	En dérivant le modèle géométrique direct proposé en \cref{app_direct_kinem_sub_3}, par chacune des articulations actionnables $ ( q_1, q_2, q_3 ) $, et en simplifiant l'écriture des cosinus et sinus respectivement par $ c $ et $ s $ avec les indices désignant les articulations - \textit{eg. $ \sin(q_1+q_2) $ est donné par $ s_{12} $} - pour une meilleure lisibilité, on obtient le résultat présenté en équation \cref{eq_jacobian_detailed}, pour $ \jacobian{}{}{} \in \R^{3 \times 3} $.
	
	\begin{equation} \label{eq_jacobian_detailed}
		\jacobian{q}{}{} = 
		\begin{pmatrix}
			-l_1 s_1 - l_2 s_{12} - l_3 s_{123} & -l_2 s_{12} - l_3 s_{123}  & -l_3 s_{123} \\
			l_1 c_1 + l_2 c_{12} + l_3 c_{123} & l_2 c_{12} + l_3 c_{123}  & l_3 c_{123} \\
			1 & 1 & 1 \\
		\end{pmatrix}
	\end{equation}
	
	\subsection{Jacobien du youBot arm} \label{app_jacobian_sub_5}
	
	On rappelle en équation \cref{eq_jacobian_joint_space}, la relation liant les vitesses articulaires aux vitesses cartésiennes à l'aide du jacobien, avec $ \vect{v} \in \R^3 $ et $ \vect{\omega} \in \R^3 $, respectivement les vitesses cartésiennes linéaires et angulaires de l'effecteur. On note $ \jacobian{}{v}{}, \in \R^{3\times n} $ et $ \jacobian{}{\omega}{} \in \R^{3\times n} $, les jacobiens linéaire et angulaire, et $ \jacobian{}{}{} \in \R^{6\times n} $ le jacobien du robot, avec $ n $, le nombre d'articulations actionnées. 
	
	\begin{equation} \label{eq_jacobian_joint_space}
		\jacobian{q}{}{} \dot{\vect{q}} = \begin{pmatrix} \vect{v} \\ \vect{\omega} \end{pmatrix} = \begin{pmatrix} \jacobian{q}{v}{} \\ \jacobian{q}{\omega}{} \end{pmatrix} \dot{\vect{q}}
	\end{equation}
	
	Les vitesses angulaires peuvent être obtenues directement en utilisant la loi de composition des vitesses, à partir des différentes matrices homogènes composant le modèle géométrique direct. Il s'agit en effet des éléments des dernières colonnes des matrices de rotation $r_{13} $, $ r_{23} $ et $ r_{33} $ de chacune des transformations (dans le cas de liaisons rotoïdes). Ainsi, dans le cas de rotation la relation pour obtenir le jacobien angulaire est donnée par l'équation \cref{eq_angular_jacobian_composition}.
	
	\begin{equation} \label{eq_angular_jacobian_composition}
		\jacobian{}{\omega}{} \dot{\vect{q}} = \sum_{i=1}^{n}\omega_i^0 = 
		\begin{pmatrix}
		\jacobian{}{\omega 1}{} & ... & \jacobian{}{\omega n}{}
		\end{pmatrix} \dot{\vect{q}} =
		\begin{pmatrix}
		\vect{z}^0_1 & ... & \vect{z}^0_n
		\end{pmatrix} \dot{\vect{q}}, 
	\end{equation}
	
	avec $ \vect{z}^0_i \in \R^3 $, les éléments de la dernière colonne de la matrice de rotation $ \matrCal{R}^0_i $ entre la référence $ 0 $ et la rotation $ i $, données en \cref{eq_angular_jacobian}.
	
	\begin{equation} \label{eq_angular_jacobian}
		\begin{split}
			\vect{z}^0_1 &=  
			\begin{pmatrix}
				0 & 0 & 1 \\
			\end{pmatrix}^t \\
			%
			\vect{z}^0_2 &=  
			\begin{pmatrix}
				s_1 & c_1 & 0 \\
			\end{pmatrix}^t \\
			%
			\vect{z}^0_3 &=  
			\begin{pmatrix}
				s_1 & c_1 & 0 \\
			\end{pmatrix}^t \\
			%
			\vect{z}^0_4 &=  
			\begin{pmatrix}
				s_1 & c_1 & 0 \\
			\end{pmatrix}^t \\
			%
			\vect{z}^0_s &=  
			\begin{pmatrix}
				\frac{s_{1-(234)} - s_{1234}}{2} & \frac{c_{1234} - c_{1-(234)}}{2} & c_{234}
			\end{pmatrix}^t \\
			%
		\end{split}
	\end{equation}
	
	Le jacobien des vitesses linéaires $ \jacobian{}{v}{} $, de son côté, est plus simplement obtenu par la dérivation partielle de la position de l'effecteur $ \vect{p}^0_s $ du robot. 
	
	\begin{equation}
		\jacobian{}{vi}{} = \frac{\partial \vect{p}^0_s}{\partial q_i} = 
		\begin{pmatrix}
			\frac{\partial x_s^0}{\partial q_i} & \frac{\partial y_s^0}{\partial q_i} & \frac{\partial z_s^0}{\partial q_i} 
		\end{pmatrix}^t,
	\end{equation}

	avec $ x^0_s $, $ y^0_s $ et $ z^0_s $ les coordonnées cartésiennes de l'effecteur du robot, qui peuvent être obtenues à partir du vecteur de translation $ \vect{t}^0_s $ contenu dans la matrice de transformation homogène $ \matrCal{T}^0_s $. Ainsi on obtient les éléments du jacobien des vitesses linéaires en équation \cref{eq_linear_jacobian}.
	
	\begin{equation} \label{eq_linear_jacobian}
		\begin{matrix}
		\begin{matrix*}[l]
			\frac{\partial x_s^0}{\partial q_1} = s_1 (l_1 s_2 - a_0 + l_2 s_{23} + l_3 s_{234}) & 
			%
			\frac{\partial y_s^0}{\partial q_1} = -c_1 (l_1 s_{2} - a_0 + l_2 s_{23} + l_3 s_{234}) \\
			%
			\frac{\partial x_s^0}{\partial q_2} = -c_1 (l_1 c_2 + l_2 c_{23} + l_3 c_{234}) &
			%
			\frac{\partial y_s^0}{\partial q_2} = -s_1 (l_1 c_{2} + l_2 s_{23} + l_3 s_{234}) \\
			%
			\frac{\partial x_s^0}{\partial q_3} = -c_1 (l_2 c_{23} + l_3 c_{234}) &
			%
			\frac{\partial y_s^0}{\partial q_3} = -s_1 (l_2 s_{23} + l_3 s_{234}) \\
			%
			\frac{\partial x_s^0}{\partial q_4} = -c_1 (l_3 c_{234}) &
			%
			\frac{\partial y_s^0}{\partial q_4} = -s_1 (l_3 s_{234}) \\
			%
			\frac{\partial x_s^0}{\partial q_5} = 0 &
			%
			\frac{\partial y_s^0}{\partial q_5} = 0 \\
		\end{matrix*} \\
		\begin{matrix*}[l]
			\frac{\partial z_s^0}{\partial q_1} = 0 \\
			%
			\frac{\partial z_s^0}{\partial q_2} = -l_1 s_2 - l_2 s_{23} - l_3 s_{234} \\
			%
			\frac{\partial z_s^0}{\partial q_3} = -l_2 s_{23} - l_3 s_{234} \\
			%
			\frac{\partial z_s^0}{\partial q_4} = -l_3 s_{234} \\
			%
			\frac{\partial z_s^0}{\partial q_5} = 0 \\
		\end{matrix*}
		\end{matrix}
	\end{equation}

	\section{Effet de couplage} \label{app_coupling_effect}
	
	On souhaite vérifier si la matrice résultant de $ \jacobian{q_d}{}{} \jacobian{q}{}{t} $ est diagonale. On considère la matrice jacobienne décrite en \cref{app_jacobian_sub_3}. On signifie par les indices $ a $, $ b $ et $ c $, les consignes articulaires $ \dot{\vect{q}}_d $, et $ 1 $, $ 2 $ et $ 3 $, les mesures articulaires $ \dot{\vect{q}} $. Pour améliorer la lisibilité, on pose les éléments fournis par le système d'équation \cref{eq_elem_coupling}, on obtient alors l'équation \cref{eq_coupling_effect_jacobian}.
	
	\begin{equation} \label{eq_elem_coupling}
		\begin{split}
			A &= (l_2 \sin(q_a + q_b) + l_3 \sin(q_a + q_b + q_c)) \\
			B &= (l_2 \sin(q_1 + q_2) + l_3 \sin(q_1 + q_2 + 3)) \\
			C &= (l_2 \sin(q_a + q_b) + l_1 \sin(q_a) + l_3 \sin(q_a + q_b + q_c)) \\
			D &= (l_2 \sin(q_1 + q_2) + l_1 \sin(q_1) + l_3 \sin(q_1 + q_2 + q_3)) \\
			E &= (l_2 \cos(q_1 + q_2) + l_3 \cos(q_1 + q_2 + q_3)) \\
			F &= (l_2 \cos(q_1 + q_2) + l_1 \cos(q_1) + l_3 \cos(q_1 + q_2 + q_3)) \\
			G &= (l_2 \cos(q_a + q_b) + l_3 \cos(q_a + q_b + q_c)) \\
			H &= (l_2 \cos(q_a + q_b) + l_1 \cos(q_a) + l_3 \cos(q_a + q_b + q_c)) \\
			I &= \sin(q_1 + q_2 + q_3) \\
			J &= \cos(q_1 + q_2 + q_3) \\
			K &= -2 l_2 \sin(q_1 + q_2) - l_1 \sin(q_1) \\
			L &=  2 l_2 \cos(q_1 + q_2) + l_1 \cos(q_1) \\
		\end{split}
	\end{equation}
	
	\begin{equation} \label{eq_coupling_effect_jacobian}
	\begin{split}
	\jacobian{q_d}{}{} \jacobian{q}{}{t} &= 
	\begin{pmatrix}
	-l_1 s_a - l_2 s_{ab} - l_3 s_{abc} & -l_2 s_{ab} - l_3 s_{abc}  & -l_3 s_{abc} \\
	l_1 c_a + l_2 c_{ab} + l_3 c_{abc} & l_2 c_{ab} + l_3 c_{abc}  & l_3 c_{abc} \\
	1 & 1 & 1 \\
	\end{pmatrix} \\&
	\begin{pmatrix}
	-l_1 s_1 - l_2 s_{12} - l_3 s_{123} & l_1 c_1 + l_2 c_{12} + l_3 c_{123} & 1 \\
	-l_2 s_{12} - l_3 s_{123} & l_2 c_{12} + l_3 c_{123}  & 1 \\
	 -l_3 s_{123} & l_3 c_{123} & 1 \\
	\end{pmatrix}	\\
	\jacobian{q_d}{}{} \jacobian{q}{}{t} &= 
	\begin{pmatrix}
		A B + C D + l_3^2 s_{abc} I &
		-E A - C F + l_3^2 s_{abc} J &
		-2 l_2 s_{ab} - l_1 s_{a} - 3 l_3 s_{abc} \\
		-G B - D H - l_3^2 c_{abc} I &
		G E + H F + l_3^2 c_{abc} J &
		2 l_2 c_{ab} + l_1 c_{a} + 3 l_3 c_{abc} \\
		K - 3 l_3 I &
		L + 3 l_3 J &
		3 \\
	\end{pmatrix}
	\end{split}
	\end{equation}
	
	Pour que la matrice soit diagonale, il faut donc que les éléments hors de la diagonale soient nuls, ce qui donne un système à 6 équations et 6 variables.
	
	\begin{equation}	
		\left.\begin{array}{r}
			-E A - C F + l_3^2 s_{abc} J = 0 \\
			-2 l_2 s_{ab} - l_1 s_{a} - 3 l_3 s_{abc} = 0 \\
			-G B - D H - l_3^2 c_{abc} I = 0 \\
			2 l_2 c_{ab} + l_1 c_{a} + 3 l_3 c_{abc}  = 0 \\
			K - 3 l_3 I = 0 \\
			L + 3 l_3 J = 0 \\
		\end{array} \right\} = \kappa(q_1, q_2, q_3, q_a, q_b, q_c)
	\end{equation}
	
	Si l'on prend les valeurs des longueurs données par KUKA (en mètre), c'est à dire : $ l_1 = 0.155$, $ l_2 = 0.135 $ et $ l_3 = 0.113 $, alors le problème n'admet aucune solution. Il n'existe donc aucune configuration pour laquelle $ \jacobian{q_d}{}{} \jacobian{q}{}{t} $ est diagonale.
