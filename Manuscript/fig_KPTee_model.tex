	\newcommand{\comp}[3][]{ % comparator command
		\node[comp] (#2) at #3 {};
		\draw (#2) -- + (45:0.25) -- + (45:-0.25);

		\ifthenelse{
			\isempty{#1}{}
		}{}
		{
			\ifthenelse{
				\equal{#1}{a}
			}{
				\node[above right] at (#2) {$-$\phantom{p}};
			}{}
			\ifthenelse{
				\equal{#1}{b}
			}{
				\node[below right] at (#2) {$-$\phantom{l}};
			}{}
		}
	}
