\subsection{Impédance mécanique}
\begin{frame}
\frametitle{\insertsubsection}
	\begin{block}{Impédance \only<2>{mécanique}}
		L'impédance \only<2>{mécanique} $H$ est le rapport entre \only<1>{un potentiel extensif $E$ (\textit{eg.} tension, force)}\only<2>{une \textbf{force} $F$} à \only<1>{un flux intensif $\phi$ (\textit{eg.} courant, vitesse)}\only<2>{un \textbf{mouvement} $v$}: %\\
		\only<1>{$$ E(t)=H(t)\phi(t) $$}\only<2>{$$ F(t)=H(t)v(t) $$}
	\end{block}
	\vspace{1cm}
	\begin{figure}
		\begin{tikzpicture}
			\node[rectangle, draw, align=center, minimum width=3cm, minimum height=1.25cm, line width=1, font=\large] (impedance) at (0,0) {\emph{Impédance}};
			\node[rectangle, draw, align=center, minimum width=3cm, minimum height=1.25cm, line width=1, font=\large] (admittance) at ($(impedance) + (4.5, 0)$) {\emph{Admittance}};
			\draw[-{Latex[length=3mm]}, line width=1] (impedance.east) -- (admittance.west)
			node[midway, above] {\only<1>{$E$}\only<2>{$F$}};
			\draw[{Latex[length=3mm]}-, line width=1] (impedance.west) -- +(-1.25,0)
			node[midway, above] {\only<1>{$\phi$}\only<2>{$v$}};		
			\draw[-{Latex[length=3mm]}, line width=1] (admittance.east) -- +(1.25,0)
			node[midway, above] {\only<1>{$\phi$}\only<2>{$v$}};
		\end{tikzpicture}
	\end{figure}

\end{frame}
