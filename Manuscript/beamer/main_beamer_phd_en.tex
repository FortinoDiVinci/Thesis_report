\subsection{Mechanical Impedance}
\begin{frame}
\frametitle{\insertsubsection}
	\begin{block}{\only<2>{Mechanical} Impedance}
		The \only<2>{mechanical} impedance $H$ relates \only<1>{an extensive potential $E$ (\textit{eg.} voltage, force)}\only<2>{a \textbf{force} $F$} to \only<1>{an intensive flux $\phi$ (\textit{eg.} current, velocity)}\only<2>{a \textbf{movement} $v$}: %\\
		\only<1>{$$ E(t)=H(t)\phi(t) $$}\only<2>{$$ F(t)=H(t)v(t) $$}
	\end{block}
	\vspace{1cm}
	\begin{figure}
		\begin{tikzpicture}
			\node[rectangle, draw, align=center, minimum width=3cm, minimum height=1.25cm, line width=1, font=\large] (impedance) at (0,0) {\emph{Impedance}};
			\node[rectangle, draw, align=center, minimum width=3cm, minimum height=1.25cm, line width=1, font=\large] (admittance) at ($(impedance) + (4.5, 0)$) {\emph{Admittance}};
			\draw[-{Latex[length=3mm]}, line width=1] (impedance.east) -- (admittance.west)
			node[midway, above] {\only<1>{$E$}\only<2>{$F$}};
			\draw[{Latex[length=3mm]}-, line width=1] (impedance.west) -- +(-1.25,0)
			node[midway, above] {\only<1>{$\phi$}\only<2>{$v$}};		
			\draw[-{Latex[length=3mm]}, line width=1] (admittance.east) -- +(1.25,0)
			node[midway, above] {\only<1>{$\phi$}\only<2>{$v$}};
		\end{tikzpicture}
	\end{figure}

\end{frame}
