		\path ($ (joint2) + \coeff*(0.135*\alpB, {0.135*(1 - \alpB)}) $) coordinate (joint3);

		\newcommand{\alpC}{0.55}
		
		\path ($ (joint3) + \coeff*(0.113*\alpC, {0.113*(1 - \alpC)}) $) coordinate (endeffector);
	
		%% end of impedance drawing
		
		\draw[-{Latex[length=3mm]}] (origin.center) --+ ($ (0, 6) $)
		node[at end, left] {$\vect{z}$};

		\draw[line width=0.5mm] (joint0.center) -- 
		node[below, midway]{$ a_0 $} (joint1.center);

		\draw[line width=0.5mm] (joint1.center) -- 
		node[left, midway]{$ l_1 $} (joint2.center);
		\fill (joint2) circle[radius=2.75pt];
		
		\draw[line width=0.5mm] (joint2.center) -- 
		node[below, midway]{$ l_2 $} (joint3.center);
		\fill (joint3) circle[radius=2.75pt];
		
		\draw[line width=0.5mm] (joint3.center) -- 
		node[above, midway, xshift=-1mm, yshift=-1mm]{$ l_3 $} (endeffector.center);

		\draw[dashed] (joint2.center) -- ($(joint3Exceed.center) + (0.5*\alpB, {0.5*(1 - \alpB)})$);
